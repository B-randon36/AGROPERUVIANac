    %% =========================================================================
    class VentaController {
        -VentaService ventaService
        +buscar(estado, desde, hasta, ordenarPor, direccion, jwt) ResponseEntity~List~VentaResponse~~
        +obtener(id, jwt) ResponseEntity~VentaResponse~
        +crear(request, jwt) ResponseEntity~VentaResponse~
        +anular(id) ResponseEntity~VentaResponse~
        +reporte(estado, desde, hasta) ResponseEntity~VentaReporte~
    }

    class VentaService {
        <<interface>>
        +buscar(estado, desde, hasta, ordenarPor, direccion, vendedorId) List~VentaResponse~
        +obtener(id, vendedorId) VentaResponse
        +crear(request, vendedorId) VentaResponse
        +anular(id) VentaResponse
        +reporte(estado, desde, hasta) VentaReporte
        +eliminar(id) void
    }

    class VentaRequest {
        -MetodoPagoVenta metodoPago
        -String clienteNombre
        -String clienteDocumento
        -Integer diasCredito
        -List~DetalleVentaRequest~ detalles
    }

    class DetalleVentaRequest {
        -Long productoId
        -Long loteId
        -Integer cantidad
    }

    class VentaResponse {
        -Long id
        -String numeroTicket
        -LocalDateTime fecha
        -EstadoVenta estado
        -MetodoPagoVenta metodoPago
        -BigDecimal subtotal
        -BigDecimal igv
        -BigDecimal total
        -String clienteNombre
        -String vendedorId
        -List~DetalleVentaResponse~ detalles
    }

    class DetalleVentaResponse {
        -Long id
        -Long productoId
        -String productoNombre
        -Long loteId
        -String codigoLote
        -Integer cantidad
        -BigDecimal precioUnitario
        -BigDecimal subtotal
    }

    class VentaReporte {
        -Long totalVentas
        -BigDecimal montoTotalVendido
        -BigDecimal ticketPromedio
        -BigDecimal igvTotalRecaudado
        -Long ventasAnuladas
    }

    VentaController ..> VentaService : invoca
    VentaController ..> VentaRequest : recibe
    VentaController ..> VentaResponse : retorna
    VentaService ..> VentaRequest : valida_y_procesa
    VentaService ..> VentaResponse : produce
    VentaService ..> VentaReporte : calcula
    VentaRequest *-- DetalleVentaRequest : contiene
    VentaResponse *-- DetalleVentaResponse : contiene
